extension Color {
    init(hex: UInt32) {self.init(.sRGB,red:Double((hex>>16)&255)/255,green:Double((hex>>8)&255)/255,blue:Double(hex&255)/255,opacity:1)}
    static let sovereignInk=Color(hex:0x292B35)
    static let sky=Color(hex:0x83CFF0)
    static let lavender=Color(hex:0xB8ACE4)
    static let coral=Color(hex:0xF28779)
    static let butter=Color(hex:0xF5CE65)
    static let sage=Color(hex:0xA9CDB4)
}
extension Capability {var color: Color {switch self {case .chat:.sky;case .voice:.lavender;case .images:.butter;case .video:.coral}}}
struct PrimaryButton: ButtonStyle {
    var coral=false
    func makeBody(configuration: Configuration) -> some View {configuration.label.font(.system(size:14,weight:.semibold)).padding(.horizontal,24).padding(.vertical,14).foregroundStyle(.white).background(coral ? Color(hex:0x88445E):.sovereignInk,in:Capsule()).opacity(configuration.isPressed ? 0.8:1).scaleEffect(configuration.isPressed ? 0.98:1)}
}
struct GlassPanel: ViewModifier {
    @Environment(\.accessibilityReduceTransparency) var reduce
    func body(content: Content)->some View {content.background {if reduce{RoundedRectangle(cornerRadius:28).fill(Color(nsColor:.windowBackgroundColor))}else{RoundedRectangle(cornerRadius:28).fill(.regularMaterial)}}.overlay(RoundedRectangle(cornerRadius:28).stroke(.white.opacity(0.55),lineWidth:1)).shadow(color:.black.opacity(0.06),radius:24,y:12)}
}
extension View {func glassPanel()->some View {modifier(GlassPanel())}}

    @Environment(\.accessibilityReduceMotion) var reduce
    var active=true
    var body: some View {TimelineView(.animation(minimumInterval:0.08,paused:reduce || !active)){ctx in let t=ctx.date.timeIntervalSinceReferenceDate;ZStack{Circle().fill(Color.sky.opacity(0.35)).blur(radius:26).frame(width:180,height:180);ForEach(0..<5){i in Ellipse().fill(LinearGradient(colors:[.sky.opacity(0.8),.lavender.opacity(0.9),.coral.opacity(0.6)],startPoint:.topLeading,endPoint:.bottomTrailing)).frame(width:105+CGFloat(i)*7,height:145).rotationEffect(.degrees(Double(i)*36+(reduce ? 0:sin(t*0.7)*14))).scaleEffect(reduce ? 1:1+sin(t*1.4+Double(i))*0.045).opacity(0.62)};Image(systemName:"sparkle").font(.system(size:36,weight:.ultraLight)).foregroundStyle(.white)}.frame(width:230,height:230)}}
}
